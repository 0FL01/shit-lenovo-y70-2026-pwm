def construct_proposed_adapter(*, pin, serials, limits, parent_pin, capacity,
                               exact_capture_path, host_uid, monotonic, sleep,
                               os_api, socket_factory):

    association = LinuxAssociation(pin=pin, limits=limits, proc=proc,
                                   monotonic=monotonic)
    make_socket = make_socket_factory(socket_factory=socket_factory,
                                      monotonic=monotonic)
    attempted = False
    def create_sink(path, deadline, *, observed_size):
        nonlocal attempted
        if attempted:
            raise RuntimeError('Sink attempt already spent')
        attempted = True  # even failed validation/construction consumes this attempt
        if path != exact_capture_path or parent_pin.uid != host_uid:
            raise RuntimeError('Private sink binding mismatch')
        factory = PrivateSinkFactory(exact_path=exact_capture_path,
            parent_pin=parent_pin, observed_size=observed_size, capacity=capacity,
            os_api=os_api, monotonic=monotonic)
        return factory(path, deadline)
    # Class remains explicitly the reviewed MODEL until production admission.

        expected_serials=serials, acquire=association.acquire_process, make_socket=make_socket,
        create_sink=create_sink, expected_destination=exact_capture_path,
        expected_uid=host_uid, sleep=sleep, monotonic=monotonic)
